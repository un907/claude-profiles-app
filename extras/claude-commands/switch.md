!`sh -c 'a="$ARGUMENTS"; s="$CLAUDE_CODE_SESSION_ID"; list() { { for f in "$HOME"/.local/bin/claude-*; do [ -f "$f" ] && grep -q "^# Generated by Claude Profiles" "$f" && basename "$f"; done; IFS=:; for d in $PATH; do for f in "$d"/claude-*; do [ -f "$f" ] && [ -x "$f" ] && basename "$f"; done; done; } 2>/dev/null | sort -u | tr "\n" " "; }; if [ -z "$a" ]; then echo "利用できるコマンド: $(list)"; echo "使い方: /switch <名前> で claude-<名前> にこのセッションを引き継ぐコマンドを作ります。"; exit 0; fi; case "$a" in *[!A-Za-z0-9._-]*) echo "名前に使えるのは英数字と . _ - だけです: $a"; exit 0;; esac; if ! command -v "claude-$a" >/dev/null 2>&1; then echo "claude-$a は PATH にありません。"; echo "利用できるコマンド: $(list)"; exit 0; fi; if [ -z "$s" ]; then echo "セッションIDを取得できませんでした。"; exit 0; fi; c="claude-$a --resume $s"; printf %s "$c" | pbcopy 2>/dev/null; echo "$c"; echo "1. このセッションを /exit で終了してください。"; echo "2. クリップボードにコピーした上のコマンドを貼り付けて実行してください。"'`
